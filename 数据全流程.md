可以按 **5 个 PR/迭代** 来补，不要一次性重构。你现在已有 `asset_snapshot`、`source_document`、`canonical_event`、`assertion` 这些表，也有 `AssetSnapshotRepository`、`AssetAnalysisService`、`DataSourceRouter` 这些入口。缺的是一层“市场结构化事实表”，以及从 crawler 到这些表的 ETL 闭环。

---

## 总体目标

把现在的链路：

```text
crawler / adapter
  → AssetAnalysisSnapshot(JSON)
  → asset_snapshot
```

升级成：

```text
crawler / adapter
  → raw payload
  → normalizer
  → structured SQL tables
  → derived asset_snapshot / dashboard / API
```

也就是：**原始数据可追溯，标准表可查询，快照只是派生结果。**

```python
    def upsert_daily_bars(self, bars: list[dict]) -> int: ...
    def insert_quote_snapshots(self, quotes: list[dict]) -> int: ...

```

PostgreSQL 可以用 `sqlalchemy.dialects.postgresql.insert(...).on_conflict_do_update()`。

示例：

```python
from sqlalchemy.dialects.postgresql import insert

class MarketDataRepository(BaseRepository):
    def upsert_daily_bars(self, bars: list[dict]) -> int:
        if not bars:
            return 0

        stmt = insert(StockDailyBarDB).values(bars)
        stmt = stmt.on_conflict_do_update(
            constraint="uq_stock_daily_bar_symbol_date_source",
            set_={
                "open": stmt.excluded.open,
                "high": stmt.excluded.high,
                "low": stmt.excluded.low,
                "close": stmt.excluded.close,
                "volume": stmt.excluded.volume,
                "amount": stmt.excluded.amount,
                "turnover": stmt.excluded.turnover,
                "raw_payload": stmt.excluded.raw_payload,
            },
        )
        self.db.execute(stmt)
        self.db.flush()
        return len(bars)
```

SQLite 测试环境可以先简化成：

```python
existing = query.first()
if existing:
    update fields
else:
    add model
```

这样先跑通，不要被数据库方言卡住。

```python
def normalize_a_share_symbol(raw_code: str) -> str:
    code = str(raw_code).strip().upper()
    code = code.replace(".SH", "").replace(".SZ", "").replace(".BJ", "")
```

# PR 4：补 ETL orchestrator

新建：

```text
core/services/market_data_ingestion_service.py
```

核心职责：

```text
fetcher 拉数据
  → normalizer 标准化
  → repository upsert
  → etl_run 记录状态
```

示例：

```python
import uuid
from datetime import date, datetime

from data_layer.crawlers.akshare.base import AkShareAdapter
from data_layer.normalizers.akshare_market import normalize_market_data, normalize_stock_info
from data_layer.repositories.market_data_repository import MarketDataRepository
from data_layer.repositories.etl_run_repository import ETLRunRepository

class MarketDataIngestionService:
    def __init__(
        self,
        market_repo: MarketDataRepository,
        etl_repo: ETLRunRepository,
        akshare_adapter: AkShareAdapter | None = None,
    ):
        self.market_repo = market_repo
        self.etl_repo = etl_repo
        self.akshare = akshare_adapter or AkShareAdapter()

    def ingest_stock_master(self, limit: int | None = None) -> dict:
        run_id = str(uuid.uuid4())
        self.etl_repo.start(run_id, job_name="ingest_stock_master", source="akshare")

        try:
            stocks = self.akshare.market.get_stock_list(limit=limit)
            rows = [normalize_stock_info(x) for x in stocks]
            saved = self.market_repo.upsert_stock_master_many(rows)

            self.etl_repo.finish(
                run_id,
                status="success",
                items_fetched=len(stocks),
                items_normalized=len(rows),
                items_saved=saved,
            )
            return {"run_id": run_id, "fetched": len(stocks), "saved": saved}
        except Exception as e:
            self.etl_repo.fail(run_id, str(e))
            raise

    def ingest_daily_bars(
        self,
        symbols: list[str],
        start_date: date,
        end_date: date,
    ) -> dict:
        run_id = str(uuid.uuid4())
        self.etl_repo.start(run_id, job_name="ingest_daily_bars", source="akshare")

        fetched = 0
        rows = []

        try:
            for symbol in symbols:
                bars = self.akshare.market.get_historical_data(
                    symbol=symbol,
                    start_date=start_date,
                    end_date=end_date,
                )
                fetched += len(bars)
                rows.extend(normalize_market_data(x) for x in bars)

            saved = self.market_repo.upsert_daily_bars(rows)

            self.etl_repo.finish(
                run_id,
                status="success",
                items_fetched=fetched,
                items_normalized=len(rows),
                items_saved=saved,
            )
            return {"run_id": run_id, "fetched": fetched, "saved": saved}
        except Exception as e:
            self.etl_repo.fail(run_id, str(e))
            raise
```
